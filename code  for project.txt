const ACS_YEAR=2024;

// Normalize city names to compare Census records
function normalizeCity(name){
return name.toLowerCase()
.trim()
.replace(/[^a-z0-9 ]/g,"")
.replace(/\b(city|town|village|borough|municipality|cdp|urban)\b/g,"")
.replace(/\s+/g," ")
.trim();
}

// Look up population through the Census API
async function getPopulation(city,state){

const stateCode=states[state.trim().toUpperCase()];

if(!stateCode){
throw new Error("Please enter a valid U.S. state name or two-letter abbreviation.");
}

const stateName=STATE_NAMES[stateCode];

const url=
`https://api.census.gov/data/${ACS_YEAR}/acs/acs5?get=NAME,B01003_001E&for=place:*&in=state:${stateCode}`;

const response=await fetch(url);

if(!response.ok){
throw new Error("The Census Bureau population service is currently unavailable.");
}

const data=await response.json();

if(!data || data.length<2){
throw new Error("No population records were returned for this state.");
}

const searchName=normalizeCity(city);

const matches=data.slice(1).filter(row=>{
const censusName=row[0].split(",")[0];
return normalizeCity(censusName)===searchName;
});

if(matches.length===0){
throw new Error(
`Could not find "${city}" as a Census-designated incorporated place or CDP in ${stateName}. Check the spelling or try the community's official Census place name.`
);
}

if(matches.length>1){
throw new Error("Multiple Census places matched this name. Please provide a more specific community name.");
}

const population=Number(matches[0][1]);

if(!Number.isFinite(population) || population<=0){
throw new Error("The Census Bureau did not provide a usable population estimate for this community.");
}

return{
population,
censusName:matches[0][0],
year:ACS_YEAR,
stateName,
source:"U.S. Census Bureau, American Community Survey 5-Year Estimates",
table:"B01003 (Total Population)",
url:`https://api.census.gov/data/${ACS_YEAR}/acs/acs5`
};
}
